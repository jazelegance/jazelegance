  <script>
    let products = [
      { id: 1, name: "حقيبة ديور الفاخرة الوردية", price: 1200, category: "women", image: "dior_pink.png" },
      { id: 2, name: "حقيبة لويس فيتون الكلاسيكية الأزرق والبيج", price: 1500, category: "women", image: "lv_bag.png" },
      { id: 3, name: "محفظة وحقيبة كتف إيف سان لوران حمراء", price: 950, category: "women", image: "ysl_red.png" },
      { id: 4, name: "حقيبة غويا الكلاسيكية الفاخرة", price: 1800, category: "women", image: "goyard.png" }
    ];

    const mockUsers = {
      "admin1": { password: "admin123", role: "admin", name: "إدارة جاز" },
      "customer1": { password: "user123", role: "customer", name: "كرم" }
    };

    let currentUser = null;
    let currentCategory = 'all';
    let cartCount = 0;

    window.onload = function() { renderProducts(); };

    function renderProducts() {
      const grid = document.getElementById('productsGrid');
      grid.innerHTML = '';
      const filtered = products.filter(p => currentCategory === 'all' || p.category === currentCategory);

      if (filtered.length === 0) {
        grid.innerHTML = `<div class="empty-notice">لا توجد منتجات متوفرة حالياً في هذا القسم.</div>`;
        return;
      }

      filtered.forEach(product => {
        const card = document.createElement('div');
        card.className = 'product-card';
        const adminDisplay = (currentUser && currentUser.role === 'admin') ? 'block' : 'none';
        const clientDisplay = (currentUser && currentUser.role === 'admin') ? 'none' : 'block';

        card.innerHTML = `
          <div class="product-image">
            ${product.image ? `<img src="${product.image}" alt="${product.name}">` : '✨'}
          </div>
          <div class="product-info">
            <h3 class="product-title">${product.name}</h3>
            <p class="price">${product.price} ر.س</p>
            <button class="product-action-btn" style="display: ${clientDisplay}" onclick="addToCart()">🛍 أضف للسلة</button>
            <div class="admin-controls" style="display: ${adminDisplay}">
              <button class="btn-admin-action" onclick="editName(${product.id})">✏️ تعديل الاسم</button>
              <button class="btn-admin-action" onclick="editPrice(${product.id})">💰 تعديل السعر</button>
              <button class="btn-admin-action" style="border-color:#ff4a4a; color:#ff4a4a;" onclick="deleteProduct(${product.id})">❌ حذف المنتج</button>
            </div>
          </div>
        `;
        grid.appendChild(card);
      });
    }

    function filterCategory(category, element) {
      currentCategory = category;
      document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
      element.classList.add('active');
      renderProducts();
    }

    function openLoginModal() { document.getElementById('loginModal').style.display = 'flex'; }
    function closeLoginModal() { document.getElementById('loginModal').style.display = 'none'; }

    function handleAuth(event) {
      event.preventDefault();
      const userInp = document.getElementById('username').value;
      const passInp = document.getElementById('password').value;
      const errorMsg = document.getElementById('loginError');

      if (mockUsers[userInp] && mockUsers[userInp].password === passInp) {
        currentUser = mockUsers[userInp];
        errorMsg.style.display = 'none';
        closeLoginModal();
        document.getElementById('loginBtn').style.display = 'none';
        document.getElementById('welcomeMsg').textContent = `مرحباً، ${currentUser.name}`;
        document.getElementById('welcomeMsg').style.display = 'inline-block';
        document.getElementById('logoutBtn').style.display = 'inline-block';
        renderProducts();
      } else {
        errorMsg.textContent = "اسم المستخدم أو كلمة المرور غير صحيحة!";
        errorMsg.style.display = 'block';
      }
    }

    function logout() {
      currentUser = null;
      document.getElementById('loginBtn').style.display = 'inline-block';
      document.getElementById('welcomeMsg').style.display = 'none';
      document.getElementById('logoutBtn').style.display = 'none';
      renderProducts();
    }

    function addToCart() {
      cartCount++;
      document.getElementById('cartCount').textContent = cartCount;
    }

    function editPrice(id) {
      const product = products.find(p => p.id === id);
      if(!product) return;
      const newPrice = prompt(`أدخل السعر الجديد لـ (${product.name}):`, product.price);
      if(newPrice && !isNaN(newPrice)) { product.price = parseFloat(newPrice); renderProducts(); }
    }

    function editName(id) {
      const product = products.find(p => p.id === id);
      if(!product) return;
      const newName = prompt(`أدخل الاسم الجديد:`, product.name);
      if(newName && newName.trim() !== "") { product.name = newName.trim(); renderProducts(); }
    }

    function deleteProduct(id) {
      if(confirm("هل تريد حذف هذا المنتج؟")) { products = products.filter(p => p.id !== id); renderProducts(); }
    }
  </script>
